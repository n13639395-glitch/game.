from pathlib import Path

content = """name: Build APK

on:
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-22.04

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Install system packages
        run: |
          sudo apt-get update
          sudo apt-get install -y git zip unzip openjdk-17-jdk autoconf automake libtool pkg-config build-essential cmake

      - name: Install p4a and Cython
        run: |
          python3 -m pip install --upgrade pip
          python3 -m pip install python-for-android
          python3 -m pip install "cython==0.29.33"

      - name: Build APK
        run: |
          p4a apk \
            --private . \
            --package org.mygame \
            --name "My Game" \
            --version 1.0 \
            --bootstrap sdl2 \
            --requirements=python3,kivy \
            --arch arm64-v8a \
            --arch armeabi-v7a

      - name: Find APK
        run: find . -name "*.apk" -type f

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: My-Game-APK
          path: "**/*.apk"
"""

path = Path("/mnt/data/build-apk.yml")
path.write_text(content, encoding="utf-8")
print(f"فایل آماده شد: {path}")
